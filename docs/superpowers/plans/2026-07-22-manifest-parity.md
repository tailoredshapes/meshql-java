# Manifest Parity (Java) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Port meshql-rs's deployment manifest (schema, generator, conformance tests, HTTP serving) to `meshql` (Java), wired into `examples/farm`.

**Architecture:** A `ManifestGenerator.generate(configDir)` reads `config/graph/*.graphql` + `config/json/*.schema.json` directly off disk (bypassing the runtime `Config` object) and emits a `manifest.schema.json`-conformant `JsonNode`. The committed `examples/farm/config/manifest.json` is regenerated by a small `GenManifest` CLI class and served at `GET /manifest` via a new `Server.init(config, extraServlets)` overload.

**Tech Stack:** Java 21, Maven, Jackson (`ObjectMapper`/`JsonNode`), `com.networknt:json-schema-validator` (already a dependency), JUnit 5.

**Spec:** `/tank/repos/tailoredshapes/meshql-rs/docs/superpowers/specs/2026-07-22-manifest-parity-design.md` (this repo is `/tank/repos/tailoredshapes/meshql`, a sibling repo — the spec lives in `meshql-rs` because that's where the manifest concept originated).

```java
package com.meshql.examples.farm;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.node.ObjectNode;

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.List;
import java.util.stream.Collectors;
import java.util.stream.Stream;

/**
 * Builds the deployment manifest (schemas/manifest.schema.json) from
 * config/graph/*.graphql and config/json/*.schema.json. Mirrors
 * meshql-rs's examples/egg-economy/src/manifest.rs: reads the config
 * directory directly (not the runtime Config object), deterministic
 * (sorted by filename), path convention by construction
 * (/{entity}/graph, /{entity}/api).
 */
public class ManifestGenerator {
    private static final ObjectMapper MAPPER = new ObjectMapper();

    public static JsonNode generate(Path configDir) throws IOException {
        ObjectNode entities = MAPPER.createObjectNode();
        Path graphDir = configDir.resolve("graph");
        Path jsonDir = configDir.resolve("json");

        List<Path> graphqlFiles;
        try (Stream<Path> paths = Files.list(graphDir)) {
            graphqlFiles = paths
                    .filter(p -> p.toString().endsWith(".graphql"))
                    .sorted()
                    .collect(Collectors.toList());
        }

        for (Path path : graphqlFiles) {
            String filename = path.getFileName().toString();
            String entity = filename.substring(0, filename.length() - ".graphql".length());
            String graphqlText = Files.readString(path);

            ObjectNode surfaces = MAPPER.createObjectNode();
            ObjectNode graphSurface = MAPPER.createObjectNode();
            graphSurface.put("kind", "graphql");
            graphSurface.put("path", "/" + entity + "/graph");
            graphSurface.put("schema", graphqlText);
            surfaces.set("graph", graphSurface);

            Path jsonSchemaPath = jsonDir.resolve(entity + ".schema.json");
            if (Files.exists(jsonSchemaPath)) {
                JsonNode jsonSchema = MAPPER.readTree(jsonSchemaPath.toFile());
                ObjectNode apiSurface = MAPPER.createObjectNode();
                apiSurface.put("kind", "rest");
                apiSurface.put("path", "/" + entity + "/api");
                apiSurface.set("schema", jsonSchema);
                surfaces.set("api", apiSurface);
            }

            ObjectNode entityNode = MAPPER.createObjectNode();
            entityNode.set("surfaces", surfaces);
            entities.set(entity, entityNode);
        }

        ObjectNode manifest = MAPPER.createObjectNode();
        manifest.put("meshql", 1);
        manifest.set("entities", entities);
        manifest.set("surfaces", MAPPER.createObjectNode());
        return manifest;
    }
}
```

```java
package com.meshql.examples.farm;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;

/**
 * Regenerates examples/farm/config/manifest.json. Run from the module
 * directory (examples/farm) so the default "config" path resolves.
 */
public class GenManifest {
    public static void main(String[] args) throws IOException {
        Path configDir = Paths.get(args.length > 0 ? args[0] : "config");
        JsonNode manifest = ManifestGenerator.generate(configDir);

        ObjectMapper mapper = new ObjectMapper();
        String output = mapper.writerWithDefaultPrettyPrinter().writeValueAsString(manifest);

        Path outputPath = configDir.resolve("manifest.json");
        Files.writeString(outputPath, output + "\n");
        System.out.println("Wrote " + outputPath.toAbsolutePath());
    }
}
```

```java
package com.meshql.examples.farm;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.networknt.schema.JsonSchema;
import com.networknt.schema.JsonSchemaFactory;
import com.networknt.schema.SpecVersion;
import com.networknt.schema.ValidationMessage;
import org.junit.jupiter.api.Test;

import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.util.List;
import java.util.Set;
import java.util.stream.Stream;

import static org.junit.jupiter.api.Assertions.*;

class ManifestConformanceTest {
    private static final ObjectMapper MAPPER = new ObjectMapper();
    private static final Path CONFIG_DIR = Paths.get("config");
    private static final Path MANIFEST_PATH = CONFIG_DIR.resolve("manifest.json");
    // Relative to examples/farm/ (the module dir, = surefire's CWD): up
    // to examples/, up to repo root, then into the vendored schema.
    private static final Path SCHEMA_PATH = Paths.get("../../schemas/manifest.schema.json");

    @Test
    void manifestValidatesAgainstPublishedSchema() throws Exception {
        JsonNode schemaNode = MAPPER.readTree(SCHEMA_PATH.toFile());
        JsonNode manifest = MAPPER.readTree(MANIFEST_PATH.toFile());

        JsonSchemaFactory factory = JsonSchemaFactory.getInstance(SpecVersion.VersionFlag.V202012);
        JsonSchema schema = factory.getSchema(schemaNode);
        Set<ValidationMessage> errors = schema.validate(manifest);

        assertTrue(errors.isEmpty(), "manifest invalid: " + errors);
    }

    @Test
    void manifestMatchesRegeneration() throws Exception {
        JsonNode committed = MAPPER.readTree(MANIFEST_PATH.toFile());
        JsonNode generated = ManifestGenerator.generate(CONFIG_DIR);

        assertEquals(committed, generated,
            "config/manifest.json is stale — regenerate (see Task 3, Step 3 of the manifest-parity plan)");
    }

    @Test
    void everyGraphEntityAppearsInManifestWithCorrectSurfaces() throws Exception {
        JsonNode manifest = MAPPER.readTree(MANIFEST_PATH.toFile());
        JsonNode entities = manifest.get("entities");

        List<Path> graphFiles;
        try (Stream<Path> paths = Files.list(CONFIG_DIR.resolve("graph"))) {
            graphFiles = paths.filter(p -> p.toString().endsWith(".graphql")).toList();
        }

        int seen = 0;
        for (Path path : graphFiles) {
            String filename = path.getFileName().toString();
            String entity = filename.substring(0, filename.length() - ".graphql".length());

            JsonNode e = entities.get(entity);
            assertNotNull(e, "entity '" + entity + "' missing from manifest");
            assertEquals("graphql", e.at("/surfaces/graph/kind").asText(), entity + " graph surface");

            boolean hasJsonSchema = Files.exists(CONFIG_DIR.resolve("json").resolve(entity + ".schema.json"));
            if (hasJsonSchema) {
                assertEquals("rest", e.at("/surfaces/api/kind").asText(), entity + " api surface");
            } else {
                assertTrue(e.get("surfaces").path("api").isMissingNode(),
                    entity + " must not advertise an api surface it doesn't have");
            }
            seen++;
        }
        assertEquals(seen, entities.size(), "manifest entity count != graph file count");
    }
}
```

```java
import com.meshql.server.Server;

import java.net.URI;
import java.net.http.HttpClient;

import java.util.HashMap;
import java.util.List;
import java.util.Map;

import static org.junit.jupiter.api.Assertions.assertEquals;

class ManifestServingIntegrationTest {
    // Fixed port: Server.java doesn't currently expose the bound port
    // when started on an ephemeral one (port 0), so this test uses a
    // high, unlikely-to-collide port rather than adding that capability
    // (out of scope for this change).
    private static final int TEST_PORT = 34567;

    @Test
    void manifestServletServesTheGivenContentAt200() throws Exception {
        String manifestJson = "{\"meshql\":1,\"entities\":{},\"surfaces\":{}}";
        Config config = new Config(List.of(), List.of(), TEST_PORT, List.of(), "*", 5000, 5000);
        Server server = new Server(new HashMap<>());

        try {
            server.init(config, Map.of("/manifest", new ManifestServlet(manifestJson)));

            HttpClient client = HttpClient.newHttpClient();
            HttpRequest request = HttpRequest.newBuilder(URI.create("http://localhost:" + TEST_PORT + "/manifest"))
                    .GET().build();
            HttpResponse<String> response = client.send(request, HttpResponse.BodyHandlers.ofString());

            assertEquals(200, response.statusCode());
            assertEquals(manifestJson, response.body());
        } finally {
            server.stop();
        }
    }
}
```

```java
package com.meshql.examples.farm;

import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServlet;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;

import java.io.IOException;

/** Serves a fixed JSON string (the committed manifest) at GET. */
public class ManifestServlet extends HttpServlet {
    private final String manifestJson;

    public ManifestServlet(String manifestJson) {
        this.manifestJson = manifestJson;
    }

    @Override
    protected void doGet(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        response.setContentType("application/json");
        response.setStatus(HttpServletResponse.SC_OK);
        response.getWriter().write(manifestJson);
    }
}
```
